find_package(Qt6 6.10 REQUIRED COMPONENTS Widgets Gui Svg DBus)

get_filename_component(project_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(qrc "<RCC><qresource prefix=\"/org/braurser/kit\">\n")
file(GLOB icons CONFIGURE_DEPENDS "${project_root}/data/icons/*.svg")
foreach(icon IN LISTS icons)
    get_filename_component(name "${icon}" NAME)
    string(APPEND qrc "<file alias=\"icons/${name}\">${icon}</file>\n")
endforeach()
file(READ "${project_root}/data/tokens.json" tokens)
file(READ "${project_root}/data/icons/braurser-chevron-down-symbolic.svg" chevron)
foreach(mode IN ITEMS light dark)
    string(APPEND qrc "<file alias=\"braurser-${mode}.qss\">${CMAKE_BINARY_DIR}/generated/theme/braurser-${mode}.qss</file>\n")
    foreach(state IN ITEMS normal disabled)
        if(state STREQUAL "normal")
            set(token br_text)
        else()
            set(token br_muted)
        endif()
        string(JSON color GET "${tokens}" color "${token}" "${mode}")
        string(REPLACE "currentColor" "${color}" rendered "${chevron}")
        set(path "${CMAKE_CURRENT_BINARY_DIR}/chevron-${mode}-${state}.svg")
        file(WRITE "${path}.new" "${rendered}")
        configure_file("${path}.new" "${path}" COPYONLY)
        string(APPEND qrc "<file alias=\"controls/chevron-${mode}-${state}.svg\">${path}</file>\n")
    endforeach()
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${project_root}/data/icons/braurser-chevron-down-symbolic.svg")
string(APPEND qrc "</qresource></RCC>\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/braurser_qt_kit.qrc.new" "${qrc}")
configure_file("${CMAKE_CURRENT_BINARY_DIR}/braurser_qt_kit.qrc.new" "${CMAKE_CURRENT_BINARY_DIR}/braurser_qt_kit.qrc" COPYONLY)

add_library(braurser-qt-kit STATIC
    theme.cpp widgets.cpp document.cpp
    "${project_root}/include/braurser/qt/kit.hpp"
    "${CMAKE_CURRENT_BINARY_DIR}/braurser_qt_kit.qrc")
set_target_properties(braurser-qt-kit PROPERTIES AUTOMOC ON AUTORCC ON AUTORCC_OPTIONS "--format-version;1")
target_include_directories(braurser-qt-kit PUBLIC "${project_root}/include" PRIVATE "${CMAKE_BINARY_DIR}/generated/theme")
target_link_libraries(braurser-qt-kit PUBLIC Qt6::Widgets Qt6::Gui Qt6::Svg Qt6::DBus)
braurser_harden(braurser-qt-kit)

add_executable(braurser-qt-gallery gallery.cpp)
target_link_libraries(braurser-qt-gallery PRIVATE braurser-qt-kit)
braurser_harden(braurser-qt-gallery)
add_subdirectory(app)
target_link_libraries(braurser-qt-gallery PRIVATE braurser-qt-app braurser-presenter-fixtures)
