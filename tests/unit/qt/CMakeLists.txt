if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.25)
    project(braurser-qt-tests LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 23)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    get_filename_component(BRAURSER_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
    include("${BRAURSER_PROJECT_ROOT}/cmake/Hardening.cmake")
    include("${BRAURSER_PROJECT_ROOT}/cmake/Theme.cmake")
    braurser_generate_theme(OUTPUT_DIR "${CMAKE_BINARY_DIR}/generated/theme")
    enable_testing()
    add_subdirectory("${BRAURSER_PROJECT_ROOT}/src/qt" "${CMAKE_BINARY_DIR}/qt")
endif()

find_package(GTest REQUIRED)
find_package(Qt6 6.10 REQUIRED COMPONENTS Test)
add_executable(braurser-qt-kit-tests kit.cpp)
target_include_directories(braurser-qt-kit-tests PRIVATE "${CMAKE_BINARY_DIR}/generated/theme")
target_link_libraries(braurser-qt-kit-tests PRIVATE braurser-qt-kit GTest::gtest Qt6::Test)
braurser_harden(braurser-qt-kit-tests)
add_test(NAME QtKit.Components COMMAND braurser-qt-kit-tests)
set_tests_properties(QtKit.Components PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_subdirectory(app)
